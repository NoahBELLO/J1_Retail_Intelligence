"""Fonctions communes : session Spark, schémas, lecture des sources, journal."""

import os
from datetime import datetime

from pyspark.sql import SparkSession, functions as F
from pyspark.sql.types import (StructType, StructField, IntegerType, StringType,
                               DecimalType, DateType)

import config


# ---------- Session Spark
def creer_session(nom):
    spark = (
        SparkSession.builder
        .appName(nom)
        .master(config.SPARK_MASTER)
        .config("spark.driver.memory", config.SPARK_MEMOIRE)
        # PySpark (installé avec pip) ne lit pas hdfs-site.xml : on impose 1 copie, comme notre HDFS à 1 DataNode
        .config("spark.hadoop.dfs.replication", config.HDFS_REPLICATION)
        .getOrCreate()
    )
    spark.sparkContext.setLogLevel("ERROR")
    return spark


# ---------- Journal : chaque script écrit ce qu'il fait dans logs/<script>.log
class Journal:
    def __init__(self, nom_script):
        os.makedirs(config.DOSSIER_LOGS, exist_ok=True)
        self.fichier = os.path.join(config.DOSSIER_LOGS, nom_script + ".log")
        with open(self.fichier, "w", encoding="utf-8") as f:
            f.write(f"=== {nom_script} - lancé le {datetime.now():%Y-%m-%d %H:%M:%S}\n")

    def ecrire(self, *message):
        texte = " ".join(str(m) for m in message)
        print(texte)
        with open(self.fichier, "a", encoding="utf-8") as f:
            f.write(texte + "\n")


# ---------- Schémas écrits à la main (noms du schéma TPCx-BB, types vérifiés aux étapes 2 et 3)
def montant(nom):

# ---------- Lecture d'un fichier .dat de la zone raw
def lire_source_raw(spark, chemin, schema):
    """Lit un fichier TPCx-BB (séparateur |, sans ligne de titres) avec un schéma imposé.
    Les lignes qui ne se convertissent pas arrivent dans la colonne _corrupt_record.
    On ajoute la provenance : fichier d'origine, lot et date du traitement."""
    df = (
        spark.read
        .option("header", False)
        .option("sep", "|")
        .option("mode", "PERMISSIVE")
        .option("columnNameOfCorruptRecord", "_corrupt_record")
        .schema(schema)
        .csv(chemin)
        .withColumn("source_fichier", F.col("_metadata.file_path"))
        .withColumn("source_lot", F.lit(config.LOT))
        .withColumn("date_traitement", F.current_timestamp())
    )
    return df.cache()

================ scripts/01_silver.py
"""Étape 6 : raw -> silver.

Lit store_sales et item dans la zone raw, applique les règles de nettoyage de l'étape 4,
écrit les données propres dans silver (Parquet) et les lignes mises à l'écart dans quality/rejets.
Relit ensuite ce qui a été écrit pour le contrôler.
"""

from pyspark.sql import functions as F
from pyspark.sql.window import Window

import config
from commun import creer_session, Journal, lire_source_raw, SCHEMA_VENTES, SCHEMA_PRODUITS

journal = Journal("01_silver")
spark = creer_session("TP3-01-Silver")

COLONNES_PROVENANCE = ["source_fichier", "source_lot", "date_traitement"]


# ---------- Lecture
ventes_brut = lire_source_raw(spark, config.RAW_VENTES, SCHEMA_VENTES)
produits_brut = lire_source_raw(spark, config.RAW_PRODUITS, SCHEMA_PRODUITS)
journal.ecrire("Lu :", config.RAW_VENTES, "->", ventes_brut.count(), "lignes")
journal.ecrire("Lu :", config.RAW_PRODUITS, "->", produits_brut.count(), "lignes")


# ---------- Règles sur les produits
def nettoyer_produits(df):
    df = df.withColumn("nb_meme_id", F.count("*").over(Window.partitionBy("i_item_sk")))
    df = df.withColumn("motif_rejet",
        F.when(F.col("_corrupt_record").isNotNull(), "type incorrect")
         .when(F.col("i_item_sk").isNull(), "identifiant manquant")
         .when(F.col("nb_meme_id") > 1, "identifiant en double")
         .when(F.col("i_category").isNull(), "catégorie manquante"))
    colonnes_texte = [c for c, t in df.dtypes
                      if t == "string" and c not in ("motif_rejet", "_corrupt_record", "source_fichier", "source_lot")]
    for c in colonnes_texte:
        df = df.withColumn(c, F.trim(F.col(c)))
    for c in ["i_size", "i_units"]:
        df = df.withColumn(c, F.when(F.col(c).isin("N/A", "Unknown", ""), None).otherwise(F.col(c)))
    propres = df.filter(F.col("motif_rejet").isNull()) \
                .drop("motif_rejet", "nb_meme_id", "_corrupt_record", "i_container")
    rejetes = df.filter(F.col("motif_rejet").isNotNull()).drop("nb_meme_id")
    return propres, rejetes


# ---------- Règles sur les ventes
def nettoyer_ventes(df, produits_propres):
    ids_produits = produits_propres.select(F.col("i_item_sk").alias("ss_item_sk"),
                                           F.lit(True).alias("produit_connu"))
    colonnes_metier = [c for c in SCHEMA_VENTES.fieldNames() if c != "_corrupt_record"]
    df = (
        df.dropDuplicates(colonnes_metier)
        .join(ids_produits, on="ss_item_sk", how="left")
        .withColumn("motif_rejet",
            F.when(F.col("_corrupt_record").isNotNull(), "type incorrect")
             .when(F.col("ss_item_sk").isNull() | F.col("ss_ticket_number").isNull()
                   | F.col("ss_sold_date_sk").isNull() | F.col("ss_store_sk").isNull(), "identifiant manquant")
             .when(F.col("produit_connu").isNull(), "produit inconnu")
             .when(F.col("ss_quantity") <= 0, "quantité <= 0")
             .when((F.col("ss_sales_price") < 0) | (F.col("ss_net_paid") < 0), "montant négatif"))
        .withColumn("prix_zero", F.col("ss_sales_price") == 0)
    )
    propres = df.filter(F.col("motif_rejet").isNull()) \
                .select(colonnes_metier + ["prix_zero"] + COLONNES_PROVENANCE)
    rejetes = df.filter(F.col("motif_rejet").isNotNull()).drop("produit_connu")
    return propres, rejetes


produits_propres, produits_rejetes = nettoyer_produits(produits_brut)
ventes_propres, ventes_rejetees = nettoyer_ventes(ventes_brut, produits_propres)


# ---------- Écriture (dossiers réservés au TP, overwrite = on remplace le résultat précédent)
def ecrire_parquet(df, chemin):
    df.write.mode("overwrite").parquet(chemin)
    journal.ecrire("Écrit :", chemin)


ecrire_parquet(produits_propres, config.SILVER_PRODUITS)
ecrire_parquet(ventes_propres, config.SILVER_VENTES)
ecrire_parquet(produits_rejetes, config.QUALITY_REJETS_PRODUITS)
ecrire_parquet(ventes_rejetees, config.QUALITY_REJETS_VENTES)


# ---------- Contrôles après écriture : on relit ce qui est dans silver
journal.ecrire("\n--- Contrôles après écriture")
for nom, chemin, df_attendu, df_brut, df_rejets in [
    ("item", config.SILVER_PRODUITS, produits_propres, produits_brut, produits_rejetes),
    ("store_sales", config.SILVER_VENTES, ventes_propres, ventes_brut, ventes_rejetees),
]:
    relu = spark.read.parquet(chemin)
    nb_relu, nb_brut, nb_rejets = relu.count(), df_brut.count(), df_rejets.count()
    journal.ecrire(f"[{nom}] relecture OK : {nb_relu} lignes, {len(relu.columns)} colonnes")
    journal.ecrire(f"[{nom}] raw {nb_brut} = silver {nb_relu} + rejets {nb_rejets} + doublons retirés {nb_brut - nb_relu - nb_rejets}")
    meme_schema = [(f.name, f.dataType) for f in relu.schema] == [(f.name, f.dataType) for f in df_attendu.schema]
    journal.ecrire(f"[{nom}] mêmes colonnes et mêmes types qu'avant l'écriture : {meme_schema}")
    journal.ecrire(f"[{nom}] schéma silver :")
    for champ in relu.schema.fields:
        journal.ecrire(f"    {champ.name:25} {champ.dataType.simpleString()}")
    journal.ecrire(f"[{nom}] provenance : {relu.select('source_fichier', 'source_lot').distinct().collect()}")

relu = spark.read.parquet(config.SILVER_VENTES)
journal.ecrire("[store_sales] ventes à prix 0 repérées (prix_zero) :", relu.filter("prix_zero").count())

spark.stop()
journal.ecrire("Fin du script 01_silver")

================ scripts/02_gold.py
"""Étape 7 : silver -> gold.

Jeu Gold « chiffre d'affaires par catégorie » (question P4 du Jour 1).
Lit les ventes et les produits dans silver, vérifie la clé, fait la jointure,
calcule les indicateurs et écrit le résultat dans gold (Parquet), puis le relit.
"""

from pyspark.sql import functions as F

import config
from commun import creer_session, Journal

journal = Journal("02_gold")
spark = creer_session("TP3-02-Gold")

ventes = spark.read.parquet(config.SILVER_VENTES)
produits = spark.read.parquet(config.SILVER_PRODUITS)
journal.ecrire("Lu :", config.SILVER_VENTES, "->", ventes.count(), "lignes")
journal.ecrire("Lu :", config.SILVER_PRODUITS, "->", produits.count(), "lignes")


# ---------- Vérifier la clé avant la jointure
nb_produits = produits.count()
nb_ids = produits.select("i_item_sk").distinct().count()
nb_sans_produit = ventes.join(produits, ventes.ss_item_sk == produits.i_item_sk, "left_anti").count()
journal.ecrire(f"i_item_sk unique : {nb_produits == nb_ids} ({nb_ids} valeurs pour {nb_produits} lignes)")
journal.ecrire(f"Ventes dont le produit est inconnu : {nb_sans_produit}")


# ---------- Jointure ventes + produits (plusieurs ventes -> un produit)
ventes_enrichies = ventes.join(
    produits.select("i_item_sk", "i_category"),
    ventes.ss_item_sk == produits.i_item_sk,
    "left",
)
nb_avant, nb_apres = ventes.count(), ventes_enrichies.count()
nb_non_rapprochees = ventes_enrichies.filter(F.col("i_item_sk").isNull()).count()
journal.ecrire(f"Jointure : {nb_avant} lignes avant, {nb_apres} après, {nb_non_rapprochees} non rapprochées")


# ---------- Agrégation : une ligne = une catégorie
ca_total = ventes.agg(F.sum("ss_net_paid")).first()[0]
ca_par_categorie = (
    ventes_enrichies
    .groupBy(F.col("i_category").alias("categorie"))
    .agg(
        F.sum("ss_net_paid").alias("chiffre_affaires"),
        F.sum("ss_quantity").alias("quantite_vendue"),
        F.count("*").alias("nombre_lignes_vente"),
        F.countDistinct("ss_item_sk").alias("nombre_produits_vendus"),
    )
    .withColumn("part_ca_pct", F.round(F.col("chiffre_affaires") / F.lit(ca_total) * 100, 2).cast("decimal(5,2)"))
    .withColumn("source", F.lit("silver/store_sales + silver/item"))
    .withColumn("date_traitement", F.current_timestamp())
    .orderBy(F.desc("chiffre_affaires"))
)


# ---------- Écriture et relecture
ca_par_categorie.write.mode("overwrite").parquet(config.GOLD_CA_CATEGORIE)
journal.ecrire("Écrit :", config.GOLD_CA_CATEGORIE)

relu = spark.read.parquet(config.GOLD_CA_CATEGORIE)
journal.ecrire(f"Relecture OK : {relu.count()} lignes, {len(relu.columns)} colonnes")
journal.ecrire("Schéma gold :")
for champ in relu.schema.fields:
    journal.ecrire(f"    {champ.name:25} {champ.dataType.simpleString()}")

lignes = relu.orderBy(F.desc("chiffre_affaires")).collect()
journal.ecrire(f"\n{'categorie':26}{'chiffre_affaires':>18}{'quantite':>12}{'lignes':>9}{'produits':>10}{'part_%':>8}")
for l in lignes:
    journal.ecrire(f"{l.categorie:26}{l.chiffre_affaires:>18}{l.quantite_vendue:>12}"
                   f"{l.nombre_lignes_vente:>9}{l.nombre_produits_vendus:>10}{l.part_ca_pct:>8}")

journal.ecrire("\nCA total silver :", ca_total, "| somme gold :", relu.agg(F.sum("chiffre_affaires")).first()[0])

spark.stop()
journal.ecrire("Fin du script 02_gold")

================ scripts/03_formats.py
"""Étape 8 : comparer CSV, JSON et Parquet sur le même jeu de données (silver store_sales).

Écrit les mêmes lignes dans les 3 formats (dossier local de test), puis compare :
taille sur le disque, types retrouvés à la relecture, temps pour calculer un total.
"""

import os
import time

from pyspark.sql import functions as F

import config
from commun import creer_session, Journal

journal = Journal("03_formats")
spark = creer_session("TP3-03-Formats")

ventes = spark.read.parquet(config.SILVER_VENTES)
nb_lignes = ventes.count()
journal.ecrire("Jeu comparé :", config.SILVER_VENTES, "->", nb_lignes, "lignes,", len(ventes.columns), "colonnes")


def taille_dossier(chemin_file):
    dossier = chemin_file.replace("file://", "")
    total = 0
    for racine, _, fichiers in os.walk(dossier):
        for f in fichiers:
            if not f.startswith(".") and not f.startswith("_"):
                total += os.path.getsize(os.path.join(racine, f))
    return total


resultats = []
for fmt in ["csv", "json", "parquet"]:
    chemin = f"{config.SORTIE_FORMATS}/{fmt}"

    # Écriture
    ecriture = ventes.write.mode("overwrite")
    if fmt == "csv":
        ecriture.option("header", True).csv(chemin)
    elif fmt == "json":
        ecriture.json(chemin)
    else:
        ecriture.parquet(chemin)

    # Relecture sans donner de schéma : quels types retrouve-t-on ?
    if fmt == "csv":
        relu = spark.read.option("header", True).csv(chemin)
    elif fmt == "json":
        relu = spark.read.json(chemin)
    else:
        relu = spark.read.parquet(chemin)
    types = dict(relu.dtypes)

    # Temps pour calculer le total d'une seule colonne
    debut = time.time()
    total = relu.agg(F.sum(F.col("ss_net_paid").cast("decimal(17,2)"))).first()[0]
    duree = time.time() - debut

    resultats.append((fmt, taille_dossier(chemin), relu.count(), types["ss_net_paid"],
                      types["ss_quantity"], types["date_traitement"], total, duree))

taille_raw = 92764700  # store_sales.dat dans raw (relevé au Jour 2)
journal.ecrire(f"\nRéférence : fichier texte d'origine store_sales.dat = {taille_raw} octets")
journal.ecrire(f"\n{'format':9}{'taille (octets)':>17}{'% du .dat':>11}{'lignes':>9}"
               f"  {'ss_net_paid':14}{'ss_quantity':13}{'date_traitement':17}{'total ss_net_paid':>20}{'temps (s)':>10}")
for fmt, taille, lignes, t_montant, t_qte, t_date, total, duree in resultats:
    journal.ecrire(f"{fmt:9}{taille:>17}{taille / taille_raw * 100:>10.1f}%{lignes:>9}"
                   f"  {t_montant:14}{t_qte:13}{t_date:17}{str(total):>20}{duree:>10.2f}")

spark.stop()
journal.ecrire("Fin du script 03_formats")

================ scripts/04_qualite.py
"""Étape 9 : contrôles qualité sur les données silver et gold.

Chaque contrôle donne un statut :
- OK        : la règle est respectée ;
- KO        : la règle n'est pas respectée, il faut corriger ;
- ATTENTION : la règle est respectée mais il reste une anomalie à connaître pour interpréter les résultats.
Le rapport est affiché, écrit dans logs/04_qualite.log et dans quality/rapport_qualite (CSV).
"""

from pyspark.sql import functions as F
from pyspark.sql.types import DecimalType, IntegerType

import config
from commun import creer_session, Journal

journal = Journal("04_qualite")
spark = creer_session("TP3-04-Qualite")

ventes = spark.read.parquet(config.SILVER_VENTES).cache()
produits = spark.read.parquet(config.SILVER_PRODUITS).cache()
gold = spark.read.parquet(config.GOLD_CA_CATEGORIE).cache()
rejets_ventes = spark.read.parquet(config.QUALITY_REJETS_VENTES)
rejets_produits = spark.read.parquet(config.QUALITY_REJETS_PRODUITS)
raw_ventes = spark.read.text(config.RAW_VENTES)
raw_produits = spark.read.text(config.RAW_PRODUITS)
magasins = spark.read.option("sep", "|").csv(f"{config.LAKE}/raw/nifi/store.dat")  # déposé par NiFi au Jour 2

rapport = []


def controle(jeu, type_controle, regle, resultat, statut):
    rapport.append((jeu, type_controle, regle, str(resultat), statut))


def ok_si(condition):
    return "OK" if condition else "KO"


def nb_nulls(df, colonne):
    return df.filter(F.col(colonne).isNull()).count()


nb_ventes, nb_produits = ventes.count(), produits.count()

# ================= silver/store_sales
jeu = "silver/store_sales"
n_raw, n_rej = raw_ventes.count(), rejets_ventes.count()
controle(jeu, "Volume", "lignes raw = lignes silver + lignes mises à l'écart",
         f"{n_raw} = {nb_ventes} + {n_rej}", ok_si(n_raw == nb_ventes + n_rej))

for c in ["ss_item_sk", "ss_ticket_number", "ss_sold_date_sk", "ss_store_sk", "ss_quantity", "ss_net_paid"]:
    n = nb_nulls(ventes, c)
    controle(jeu, "Complétude", f"{c} non nul", f"{n} null", ok_si(n == 0))

for c in ["ss_promo_sk", "ss_addr_sk"]:
    n = nb_nulls(ventes, c)
    controle(jeu, "Complétude", f"{c} (non obligatoire) : part de null",
             f"{n} null ({n / nb_ventes * 100:.2f} %)", "OK" if n / nb_ventes < 0.01 else "ATTENTION")

colonnes_metier = [c for c in ventes.columns if c not in ("prix_zero", "source_fichier", "source_lot", "date_traitement")]
n_dist = ventes.dropDuplicates(colonnes_metier).count()
controle(jeu, "Unicité", "aucune ligne identique sur toutes les colonnes",
         f"{nb_ventes - n_dist} doublon(s)", ok_si(nb_ventes == n_dist))

types = dict((f.name, f.dataType) for f in ventes.schema.fields)
controle(jeu, "Validité", "montants en decimal(7,2)",
         types["ss_net_paid"].simpleString(),
         ok_si(all(types[c] == DecimalType(7, 2) for c in colonnes_metier if c.startswith("ss_") and isinstance(types[c], DecimalType))
               and types["ss_net_paid"] == DecimalType(7, 2)))
controle(jeu, "Validité", "identifiants et quantité en entier",
         types["ss_item_sk"].simpleString(),
         ok_si(all(isinstance(types[c], IntegerType) for c in colonnes_metier if c.endswith("_sk") or c in ("ss_ticket_number", "ss_quantity"))))

n = ventes.filter(F.col("ss_quantity") <= 0).count()
controle(jeu, "Cohérence", "ss_quantity > 0", f"{n} ligne(s) hors règle", ok_si(n == 0))
n = ventes.filter(F.col("ss_net_paid") < 0).count()
controle(jeu, "Cohérence", "ss_net_paid >= 0", f"{n} ligne(s) hors règle", ok_si(n == 0))
n = ventes.filter(F.col("ss_ext_sales_price") != F.col("ss_quantity") * F.col("ss_sales_price")).count()
controle(jeu, "Cohérence", "ss_ext_sales_price = ss_quantity x ss_sales_price", f"{n} ligne(s) hors règle", ok_si(n == 0))
n = ventes.filter(F.col("ss_sales_price") > F.col("ss_list_price")).count()
controle(jeu, "Cohérence", "prix de vente <= prix catalogue", f"{n} ligne(s) hors règle", ok_si(n == 0))
n = ventes.filter("prix_zero").count()
controle(jeu, "Cohérence", "ventes à prix 0 (gardées, repérées par prix_zero)", f"{n} ligne(s)", "ATTENTION" if n else "OK")
n = ventes.filter(F.col("ss_net_profit") < 0).count()
controle(jeu, "Cohérence", "ventes à marge négative (gardées)", f"{n} ligne(s) ({n / nb_ventes * 100:.1f} %)", "ATTENTION" if n else "OK")

n = ventes.join(produits, ventes.ss_item_sk == produits.i_item_sk, "left_anti").count()
controle(jeu, "Intégrité", "chaque ss_item_sk existe dans silver/item", f"{n} vente(s) sans produit", ok_si(n == 0))
ids_magasins = magasins.select(F.col("_c0").cast("int").alias("s_store_sk"))
n = ventes.join(ids_magasins, ventes.ss_store_sk == ids_magasins.s_store_sk, "left_anti").count()
controle(jeu, "Intégrité", "chaque ss_store_sk existe dans raw/nifi/store.dat", f"{n} vente(s) sans magasin", ok_si(n == 0))

# ================= silver/item
jeu = "silver/item"
n_raw, n_rej = raw_produits.count(), rejets_produits.count()
controle(jeu, "Volume", "lignes raw = lignes silver + lignes mises à l'écart",
         f"{n_raw} = {nb_produits} + {n_rej}", ok_si(n_raw == nb_produits + n_rej))
n_dist = produits.select("i_item_sk").distinct().count()
controle(jeu, "Clé candidate", "i_item_sk unique", f"{n_dist} valeurs pour {nb_produits} lignes", ok_si(n_dist == nb_produits))
for c in ["i_item_sk", "i_category"]:
    n = nb_nulls(produits, c)
    controle(jeu, "Complétude", f"{c} non nul", f"{n} null", ok_si(n == 0))
n = nb_nulls(produits, "i_size")
controle(jeu, "Complétude", "i_size (non obligatoire) : part de null", f"{n} null ({n / nb_produits * 100:.1f} %)", "ATTENTION")
types = dict((f.name, f.dataType) for f in produits.schema.fields)
controle(jeu, "Validité", "i_current_price en decimal(7,2), i_rec_start_date en date",
         f"{types['i_current_price'].simpleString()}, {types['i_rec_start_date'].simpleString()}",
         ok_si(types["i_current_price"] == DecimalType(7, 2) and types["i_rec_start_date"].simpleString() == "date"))
n = produits.filter(F.col("i_current_price") <= 0).count()
controle(jeu, "Cohérence", "i_current_price > 0", f"{n} ligne(s) hors règle", ok_si(n == 0))
n = produits.filter(F.col("i_wholesale_cost") > F.col("i_current_price")).count()
controle(jeu, "Cohérence", "coût d'achat > prix de vente (gardés)", f"{n} produit(s) ({n / nb_produits * 100:.1f} %)", "ATTENTION" if n else "OK")
n = produits.select("i_category").distinct().count()
controle(jeu, "Cohérence", "nombre de catégories = 9 (i_category_id de 1 à 9)", n, ok_si(n == 9))

# ================= gold/ca_par_categorie
jeu = "gold/ca_par_categorie"
nb_gold = gold.count()
controle(jeu, "Volume", "une ligne par catégorie de silver/item", f"{nb_gold} lignes", ok_si(nb_gold == produits.select("i_category").distinct().count()))
n = gold.select("categorie").distinct().count()
controle(jeu, "Unicité", "categorie unique", f"{n} valeurs pour {nb_gold} lignes", ok_si(n == nb_gold))
n = nb_nulls(gold, "categorie")
controle(jeu, "Complétude", "categorie non nulle", f"{n} null", ok_si(n == 0))
ca_silver = ventes.agg(F.sum("ss_net_paid")).first()[0]
ca_gold = gold.agg(F.sum("chiffre_affaires")).first()[0]
controle(jeu, "Cohérence", "somme du CA gold = CA total silver", f"{ca_gold} = {ca_silver}", ok_si(ca_gold == ca_silver))
lignes_gold = gold.agg(F.sum("nombre_lignes_vente")).first()[0]
controle(jeu, "Cohérence", "somme des lignes gold = lignes silver", f"{lignes_gold} = {nb_ventes}", ok_si(lignes_gold == nb_ventes))
somme_parts = gold.agg(F.sum("part_ca_pct")).first()[0]
controle(jeu, "Cohérence", "somme des parts = 100 % (à 0,05 près, à cause des arrondis)", f"{somme_parts} %", ok_si(abs(float(somme_parts) - 100) <= 0.05))
n = gold.filter(F.col("chiffre_affaires") < 0).count()
controle(jeu, "Cohérence", "chiffre_affaires >= 0", f"{n} ligne(s) hors règle", ok_si(n == 0))


# ---------- Affichage et écriture du rapport
journal.ecrire(f"{'Jeu':24}{'Contrôle':15}{'Règle':62}{'Résultat':38}Statut")
for jeu, type_controle, regle, resultat, statut in rapport:
    journal.ecrire(f"{jeu:24}{type_controle:15}{regle:62}{resultat:38}{statut}")

bilan = {s: sum(1 for r in rapport if r[4] == s) for s in ["OK", "ATTENTION", "KO"]}
journal.ecrire(f"\nBilan : {len(rapport)} contrôles - {bilan['OK']} OK, {bilan['ATTENTION']} ATTENTION, {bilan['KO']} KO")

df_rapport = spark.createDataFrame(rapport, ["jeu", "controle", "regle", "resultat", "statut"]) \
                  .withColumn("date_controle", F.current_timestamp())
df_rapport.coalesce(1).write.mode("overwrite").option("header", True).csv(config.QUALITY_RAPPORT)
journal.ecrire("Écrit :", config.QUALITY_RAPPORT)

journal.ecrire("Fin du script 04_qualite")
